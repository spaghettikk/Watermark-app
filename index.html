// ฟังก์ชันลบพื้นหลังที่เสถียร ไม่หลุด ไม่ติด CORS async function processAIRemoveBg() { if (uploadedPhotos.length === 0 || !uploadedPhotos[currentPhotoIndex]) { alert('กรุณาอัปโหลดรูปภาพก่อนใช้งาน ลบพื้นหลัง'); return; } const overlay = document.getElementById('aiLoadingOverlay'); overlay.classList.remove('hidden'); setTimeout(() => { try { const currentImg = uploadedPhotos[currentPhotoIndex]; const w = currentImg.naturalWidth || currentImg.width; const h = currentImg.naturalHeight || currentImg.height; const tempCanvas = document.createElement('canvas'); tempCanvas.width = w; tempCanvas.height = h; const tCtx = tempCanvas.getContext('2d'); tCtx.drawImage(currentImg, 0, 0); const imgData = tCtx.getImageData(0, 0, w, h); const data = imgData.data; // สุ่มเก็บสีมุมภาพทั้ง 4 ด้านเพื่อวิเคราะห์สีพื้นหลัง (Background Thresholding) const bgR = (data[0] + data[(w - 1) * 4] + data[(h - 1) * w * 4]) / 3; const bgG = (data[1] + data[(w - 1) * 4 + 1] + data[(h - 1) * w * 4 + 1]) / 3; const bgB = (data[2] + data[(w - 1) * 4 + 2] + data[(h - 1) * w * 4 + 2]) / 3; const tolerance = 50; // ค่าความโปร่งแสงเฉลี่ยของพื้นหลัง for (let i = 0; i < data.length; i += 4) { const r = data[i]; const g = data[i + 1]; const b = data[i + 2]; const diff = Math.sqrt((r - bgR) ** 2 + (g - bgG) ** 2 + (b - bgB) ** 2); if (diff < tolerance) { data[i + 3] = 0; // ปรับ Alpha เป็น 0 (โปร่งใส) } } tCtx.putImageData(imgData, 0, 0); const newImg = new Image(); newImg.onload = () => { uploadedPhotos[currentPhotoIndex] = newImg; renderPhotoThumbnails(); renderCanvas(); overlay.classList.add('hidden'); }; newImg.src = tempCanvas.toDataURL('image/png'); } catch (err) { overlay.classList.add('hidden'); alert('ไม่สามารถประมวลผลได้ กรุณาลองใช้อีกครั้ง'); console.error(err); } }, 100); }
